! Users
! Use 'secret' (not 'password') — stored as bcrypt, not reversible ciphertext
! brian logs in at privilege 1; 'enable' + enable-password elevates to #

! SSH
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 15
 login local
 transport input ssh

! SSH public key
! Strip 'ssh-rsa ' prefix and trailing comment — base64 body only
! Split at 254-character boundary across the two key-string lines
ip ssh pubkey-chain
 username brian
  key-string
   <public-key-string-line-1>
   <public-key-string-line-2>
  exit
 exit

! Enable RESTCONF/NETCONF (required for Terraform)
